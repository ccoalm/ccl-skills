## 做法

原来的写法有三个问题：

- `JSON.parse` 返回的是 `any`，`as User` 等于跳过检查。
- `localStorage` 里的值可能缺失、被改过，或者是旧版本留下的。
- `catch (err)` 里的 `err` 可能是任何值，不一定是 `Error`。

统一的改法是：外部来的值先当作 `unknown`，在入口处解析一次，之后的代码只拿到已经确认过结构的类型。

**2. 解析工具和请求体（`src/boundary/parse.ts`）**

下面的代码用 zod 写。仓库里如果已有别的 schema 库，就用那个，不要再加第二个。

```ts
import { z } from 'zod';
import { errorMessage } from './errors';

export type ParseResult<T> =
  | { ok: true; value: T }
  | { ok: false; error: string };

export function parseJsonText(text: string): ParseResult<unknown> {
  try {
    const value: unknown = JSON.parse(text); // 先把 any 收成 unknown
    return { ok: true, value };
  } catch (e) {
    return { ok: false, error: `invalid JSON: ${errorMessage(e)}` };
  }
}

export function parseWith<T>(schema: z.ZodType<T>, input: unknown): ParseResult<T> {
  const r = schema.safeParse(input);
  if (r.success) return { ok: true, value: r.data };
  // path 里可能有 symbol，直接 join 会抛错，所以先 map(String)
  const error = r.error.issues.map((i) => `${i.path.map(String).join('.')}: ${i.message}`).join('; ');
  return { ok: false, error };
}

// 类型从 schema 推导，不再单独手写一份 interface User
export const UserSchema = z.strictObject({   // 请求体有多余字段就拒绝
  id: z.string().min(1),
  name: z.string().min(1).max(100),
  email: z.string().email(),
});
export type User = z.infer<typeof UserSchema>;

export function parseUserBody(body: string): ParseResult<User> {
  const json = parseJsonText(body);
  return json.ok ? parseWith(UserSchema, json.value) : json;
}
```

调用方：

```ts
const parsed = parseUserBody(body);
if (!parsed.ok) return res.status(400).json({ error: parsed.error });
const user = parsed.value; // 这里的类型是 User，并且运行时已经检查过
```

**安全提示：** 如果 `User` 里有 `role`、`isAdmin`、`tenantId` 这类权限相关字段，不要从请求体读取。这些值要从认证后的会话或服务端状态得到。请求体里可能被伪造，伪造后的影响是越权。上面用了 `strictObject`，请求体里带 `role` 会直接被拒绝，测试里有对应的用例。

**3. 本地配置（`src/config.ts`）**

```ts
import { z } from 'zod';
import { parseJsonText } from './boundary/parse';

const CONFIG_KEY = 'cfg';

// 配置用 z.object，未知字段会被丢掉，这样新旧版本的客户端能互相兼容
export const ConfigSchema = z.object({
  version: z.literal(1),
  theme: z.enum(['light', 'dark']),
  pageSize: z.number().int().min(1).max(200),
});
export type Config = z.infer<typeof ConfigSchema>;

const DEFAULT_CONFIG = { version: 1, theme: 'light', pageSize: 20 } satisfies Config;

export type ConfigLoad =
  | { source: 'stored'; config: Config }
  | { source: 'default'; config: Config; reason: 'missing' | 'unreadable' | 'invalid' };

const fallback = (reason: 'missing' | 'unreadable' | 'invalid'): ConfigLoad =>
  ({ source: 'default', config: { ...DEFAULT_CONFIG }, reason }); // 返回副本，防止调用方改到默认值

// 传入函数而不是直接读 localStorage：在沙箱 iframe 或禁用 cookie 时，
// 光是访问 window.localStorage 就可能抛 SecurityError
export function loadConfig(
  getStorage: () => Pick<Storage, 'getItem'> = () => window.localStorage,
): ConfigLoad {
  let raw: string | null;
  try {
    raw = getStorage().getItem(CONFIG_KEY);
  } catch {
    return fallback('unreadable');
  }
  if (raw === null) return fallback('missing');
  const json = parseJsonText(raw);
  if (!json.ok) return fallback('invalid');
  const cfg = ConfigSchema.safeParse(json.value);
  return cfg.success ? { source: 'stored', config: cfg.data } : fallback('invalid');
}
```

以后配置结构变了，就新增 `version: 2`，再写一个从 v1 迁移的函数，不要放宽 schema 来兼容旧数据。

**4. 错误处理（`src/boundary/errors.ts`）**

```ts
function safeString(v: unknown): string {
  try {
    if (typeof v === 'string') return v;
    const s = JSON.stringify(v) as string | undefined; // undefined 或 symbol 会得到 undefined
    return s ?? String(v);
  } catch {
    try { return String(v); } catch { return '[unprintable thrown value]'; } // 无原型且有循环引用的对象两者都会抛错
  }
}

export function toError(e: unknown): Error {
  if (e instanceof Error) return e;
  if (typeof e === 'object' && e !== null && 'message' in e && typeof e.message === 'string') {
    return new Error(e.message, { cause: e }); // 兼容 iframe 等其他环境的 Error 和类 Error 对象
  }
  return new Error(safeString(e), { cause: e });
}

export const errorMessage = (e: unknown): string => toError(e).message;
```

使用：`catch (err) { log.error(errorMessage(err)); }`。如果要往上抛，用 `throw toError(err)`。

import { describe, it, expect } from 'vitest';
import { loadConfig } from './config';

const store = (raw: string | null) => () => ({ getItem: () => raw });
const good = { version: 1, theme: 'dark', pageSize: 50 };

describe('loadConfig', () => {
  it('reads a valid stored config', () => {
    expect(loadConfig(store(JSON.stringify(good)))).toEqual({ source: 'stored', config: good });

  it('defaults when missing', () => {
    expect(loadConfig(store(null))).toMatchObject({ source: 'default', reason: 'missing' });
  });
  it('defaults when storage throws', () => {
    const throwing = () => { throw new DOMException('denied', 'SecurityError'); };
    expect(loadConfig(throwing)).toMatchObject({ source: 'default', reason: 'unreadable' });
  });
  it.each([
    ['bad json', '{'],
    ['bad enum', JSON.stringify({ ...good, theme: 'neon' })],
    ['out of range', JSON.stringify({ ...good, pageSize: 0 })],
    ['old version', JSON.stringify({ ...good, version: 0 })],
  ])('defaults when %s', (_l, raw) => {
    expect(loadConfig(store(raw))).toMatchObject({ source: 'default', reason: 'invalid' });

    expect(({} as Record<string, unknown>).polluted).toBeUndefined();

  it('returns an independent default copy', () => {
    const a = loadConfig(store(null));
    a.config.pageSize = 999;
    expect(loadConfig(store(null)).config.pageSize).toBe(20);
  });

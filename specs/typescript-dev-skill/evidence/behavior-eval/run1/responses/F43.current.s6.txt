## 思路

三处问题其实是同一个：**外部来的数据只能当 `unknown`，校验通过后才算有类型**。`as` 只是让编译器不报错，运行时什么也没检查。

| 现在的写法 | 问题 | 改成 |
|---|---|---|
| `JSON.parse(body) as User` | `JSON.parse` 返回 `any`，`as` 不做运行时检查 | 用 schema 校验，类型由 schema 推导 |
| `localStorage.getItem('cfg') as Config` | 值可能是 `null`、坏掉的 JSON、旧版本格式；`getItem` 本身也可能抛错 | 校验失败就逐字段回退到默认值 |
| `catch (err) { err.message }` | 抛出的可能是字符串、`null`，或跨 realm 的 Error | 用 `unknown` 接住，再用 `toError()` 转成 Error |

最后用 tsconfig 和 ESLint 把规则卡住，防止以后又写回 `as`。

## 实现

### `src/result.ts` / `src/errors.ts`

```ts
// result.ts
export type Result<T, E = Error> = { ok: true; value: T } | { ok: false; error: E };

// errors.ts
function isErrorLike(e: unknown): e is { message: string } {
  return typeof e === 'object' && e !== null && 'message' in e && typeof e.message === 'string';
}

function safeStringify(v: unknown): string {
  try {
    return JSON.stringify(v) ?? String(v);
  } catch {
    try { return String(v); } catch { return '[unserializable]'; } // Object.create(null) 用 String() 会抛错
  }
}

export function toError(e: unknown): Error {
  if (e instanceof Error) return e;
  if (isErrorLike(e)) return new Error(e.message, { cause: e }); // 跨 iframe / vm 时 instanceof 不成立
  if (typeof e === 'string') return new Error(e);
  return new Error(`Non-Error thrown: ${safeStringify(e)}`, { cause: e });
}

export const errorMessage = (e: unknown): string => toError(e).message;
```

### `src/safe-json.ts`：全项目只有这里可以调用 `JSON.parse`

```ts
import { z } from 'zod';
import { toError } from './errors';
import type { Result } from './result';

export function parseJson(text: string): Result<unknown> {
  try {
    const value: unknown = JSON.parse(text);
    return { ok: true, value };
  } catch (e) {
    return { ok: false, error: toError(e) };
  }
}

export function parseWith<S extends z.ZodType>(schema: S, text: string): Result<z.output<S>> {
  const raw = parseJson(text);
  if (!raw.ok) return raw;
  const r = schema.safeParse(raw.value);
  return r.success ? { ok: true, value: r.data } : { ok: false, error: r.error };
}
```

### `src/user.ts`：类型从 schema 推导，两边不会对不上

```ts
import { z } from 'zod';

export const UserSchema = z.object({
  id: z.string().min(1),
  name: z.string(),
  email: z.email(),
  age: z.number().int().nonnegative().optional(),
}); // z.object 默认会丢掉未声明的字段
export type User = z.infer<typeof UserSchema>;

// 调用处
const parsed = parseWith(UserSchema, body);
if (!parsed.ok) return res.status(400).json({ error: 'invalid body' }); // 不要把 zod 的错误细节原样返回给客户端
const user = parsed.value; // 类型是 User
```

> 关于安全：`id`、`role`、租户这类决定身份或权限的字段，应该从认证信息或服务端状态里取，不能信请求体。schema 只能保证数据形状正确，不能保证数据可信。所以 `UserSchema` 不应该包含 `role`，下面的测试也专门覆盖了"多传字段会被丢掉"。

### `src/config.ts`

```ts
import { z } from 'zod';
import { parseWith } from './safe-json';
import { toError } from './errors';
import type { Result } from './result';

const KEY = 'cfg';

// 每个字段单独 .catch：一个字段坏了只回退这一个，不会把整份配置清空
const ConfigSchema = z.object({
  theme: z.enum(['light', 'dark']).catch('light'),
  pageSize: z.number().int().min(1).max(200).catch(20),
  features: z.record(z.string(), z.boolean()).catch({}),
});
export type Config = z.output<typeof ConfigSchema>;

const defaults = (): Config => ConfigSchema.parse({}); // 每次新建一份，调用方改了也不会影响别人

export function readConfig(storage: Pick<Storage, 'getItem'> = localStorage): Config {
  let text: string | null;
  try {
    text = storage.getItem(KEY); // 存储被禁用或隐私模式下可能抛 SecurityError
  } catch {
    return defaults();
  }
  if (text === null) return defaults();
  const r = parseWith(ConfigSchema, text); // 顶层不是对象时才会失败
  return r.ok ? r.value : defaults();
}

export function writeConfig(cfg: Config, storage: Pick<Storage, 'setItem'> = localStorage): Result<void> {
  try {
    storage.setItem(KEY, JSON.stringify(ConfigSchema.parse(cfg)));
    return { ok: true, value: undefined };
  } catch (e) {
    return { ok: false, error: toError(e) }; // 例如 QuotaExceededError
  }
}
```

### catch 的写法

```ts
try { ... } catch (e) {                  // strict 模式下 e 是 unknown
  logger.error(errorMessage(e));
}
promise.catch((e: unknown) => logger.error(errorMessage(e))); // .catch 回调的参数默认是 any，要手动标成 unknown
```

```ts
// safe-json.test.ts
import { describe, it, expect, expectTypeOf } from 'vitest';
import { ZodError } from 'zod';
import { parseJson, parseWith } from './safe-json';
import { UserSchema, type User } from './user';

const valid = { id: 'u1', name: 'A', email: 'a@example.com' };

describe('parseWith(UserSchema)', () => {
  it('合法输入', () => {
    const r = parseWith(UserSchema, JSON.stringify(valid));
    expect(r).toEqual({ ok: true, value: valid });
    if (r.ok) expectTypeOf(r.value).toEqualTypeOf<User>();
  });

  it('JSON 语法错误返回 SyntaxError，不抛出', () => {
    const r = parseWith(UserSchema, '{');
    expect(r.ok).toBe(false);
    if (!r.ok) expect(r.error).toBeInstanceOf(SyntaxError);
  });

  it.each([
    ['字段类型错', { ...valid, id: 1 }, ['id']],
    ['缺必填字段', { name: 'A', email: 'a@example.com' }, ['id']],
    ['邮箱格式错', { ...valid, email: 'x' }, ['email']],
  ])('%s → ZodError，path 指向出错字段', (_, input, path) => {
    const r = parseWith(UserSchema, JSON.stringify(input));
    expect(r.ok).toBe(false);
    if (!r.ok) {
      expect(r.error).toBeInstanceOf(ZodError);
      expect((r.error as ZodError).issues[0]?.path).toEqual(path);
    }
  });

  it.each(['null', '[]', '"str"', '42'])('顶层不是对象（%s）→ 失败', (text) => {
    expect(parseWith(UserSchema, text).ok).toBe(false);
  });

  it('伪造字段 role 会被丢掉（越权负向用例）', () => {
    const r = parseWith(UserSchema, JSON.stringify({ ...valid, role: 'admin' }));
    expect(r.ok && r.value).not.toHaveProperty('role');
  });

  it('__proto__ 注入不会污染原型', () => {
    const r = parseWith(UserSchema, `{"id":"u1","name":"A","email":"a@example.com","__proto__":{"isAdmin":true}}`);
    expect(r.ok).toBe(true);
    expect(Object.prototype).not.toHaveProperty('isAdmin');
    expect(r.ok && Object.keys(r.value)).toEqual(['id', 'name', 'email']);
  });

  it('parseJson 返回 unknown', () => {
    const r = parseJson('1');
    if (r.ok) expectTypeOf(r.value).toBeUnknown();
  });
});
```

```ts
// config.test.ts
import { describe, it, expect } from 'vitest';
import { readConfig, writeConfig } from './config';

const store = (v: string | null) => ({ getItem: () => v });
const DEFAULTS = { theme: 'light', pageSize: 20, features: {} };

describe('readConfig', () => {
  it.each([
    ['未设置', null],
    ['JSON 坏了', '{oops'],
    ['顶层是数字', '42'],
    ['顶层是 null', 'null'],
    ['顶层是数组', '[]'],
  ])('%s → 默认值', (_, raw) => {
    expect(readConfig(store(raw))).toEqual(DEFAULTS);
  });

  it('坏字段只回退该字段，好字段保留', () => {
    const cfg = readConfig(store('{"theme":"neon","pageSize":50,"features":{"x":true}}'));
    expect(cfg).toEqual({ theme: 'light', pageSize: 50, features: { x: true } });
  });

  it('越界值回退', () => {
    expect(readConfig(store('{"pageSize":99999}')).pageSize).toBe(20);
  });

  it('getItem 抛错 → 默认值', () => {
    const s = { getItem: () => { throw new DOMException('denied', 'SecurityError'); } };
    expect(readConfig(s)).toEqual(DEFAULTS);
  });

  it('返回的默认值不共享引用', () => {
    const a = readConfig(store(null));
    a.features.hacked = true;
    expect(readConfig(store(null)).features).toEqual({});
  });
});

describe('writeConfig', () => {
  it('超出配额返回错误，不抛出', () => {
    const s = { setItem: () => { throw new DOMException('full', 'QuotaExceededError'); } };
    const r = writeConfig({ theme: 'dark', pageSize: 10, features: {} }, s);
    expect(r.ok).toBe(false);
    if (!r.ok) expect(r.error.name).toBe('QuotaExceededError');
  });

  it('写入后能原样读回', () => {
    let saved: string | null = null;
    const cfg = { theme: 'dark' as const, pageSize: 10, features: { a: false } };
    writeConfig(cfg, { setItem: (_k, v) => { saved = v; } });
    expect(readConfig(store(saved))).toEqual(cfg);
  });
});
```

## 2. 实现（使用 zod，类型从 schema 推导）

```ts
// src/safe/errors.ts
export function getErrorMessage(err: unknown): string {
  if (err instanceof Error) return err.message;
  if (typeof err === 'string') return err;
  // 跨 realm（iframe/worker）抛出的 Error 不满足 instanceof，所以再按结构判断一次
  if (typeof err === 'object' && err !== null && 'message' in err && typeof err.message === 'string') {
    return err.message;
  }
  try {
    return JSON.stringify(err) ?? safeString(err);   // undefined/symbol 时 stringify 返回 undefined
  } catch {
    return safeString(err);                           // 循环引用、bigint
  }
}

function safeString(v: unknown): string {
  try { return String(v); } catch { return 'Unknown error'; } // Object.create(null) 时 String 会抛错
}
```

```ts
// src/safe/json.ts
import { z } from 'zod';
import { getErrorMessage } from './errors';

export type Result<T, E = string> = { ok: true; value: T } | { ok: false; error: E };

export function safeJsonParse(text: string): Result<unknown> {
  try {
    const value: unknown = JSON.parse(text); // 用类型标注而不是 as，any 在这里就收成 unknown
    return { ok: true, value };
  } catch (err) {
    return { ok: false, error: `invalid JSON: ${getErrorMessage(err)}` };
  }
}

export function parseWith<S extends z.ZodTypeAny>(schema: S, text: string): Result<z.infer<S>> {
  const json = safeJsonParse(text);
  if (!json.ok) return json;
  const r = schema.safeParse(json.value);
  if (r.success) return { ok: true, value: r.data };
  const error = r.error.issues
    .map((i) => `${i.path.map(String).join('.') || '(root)'}: ${i.message}`)
    .join('; ');
  return { ok: false, error };
}
```

```ts
// src/config.ts
import { z } from 'zod';
import { parseWith } from './safe/json';
import { getErrorMessage } from './safe/errors';

export const CONFIG_KEY = 'cfg';
export const ConfigSchema = z.object({
  theme: z.enum(['light', 'dark']).default('light'),
  pageSize: z.number().int().min(1).max(200).default(20),
  features: z.array(z.string()).default([]),
});
export type Config = z.infer<typeof ConfigSchema>;
export const defaultConfig = (): Config => ConfigSchema.parse({}); // 每次返回新对象，避免多处共享同一个默认值被改掉

type ReadStorage = Pick<Storage, 'getItem'>;

function browserStorage(): ReadStorage | undefined {
  try { return globalThis.localStorage; } catch { return undefined; } // 隐私模式或禁用存储时访问会直接抛错
}

export function loadConfig(
  storage: ReadStorage | undefined = browserStorage(),
  onInvalid?: (reason: string) => void,
): Config {
  let raw: string | null;
  try {
    raw = storage?.getItem(CONFIG_KEY) ?? null;
  } catch (err) {
    onInvalid?.(`storage unavailable: ${getErrorMessage(err)}`);
    return defaultConfig();
  }
  if (raw === null) return defaultConfig();
  const r = parseWith(ConfigSchema, raw);
  if (!r.ok) {
    onInvalid?.(r.error);
    return defaultConfig();
  }
  return r.value;
}

export function saveConfig(cfg: Config, storage: Pick<Storage, 'setItem'> = localStorage): void {
  storage.setItem(CONFIG_KEY, JSON.stringify(ConfigSchema.parse(cfg))); // 写入前也校验，坏数据不落盘
}
```

`catch` 的写法：

```ts
try { await save(); }
catch (err) {                       // err: unknown
  if (err instanceof HttpError) { /* 需要按类型分支时用 instanceof 收窄 */ }
  logger.error('save failed', { message: getErrorMessage(err), cause: err });
  throw new Error(`save failed: ${getErrorMessage(err)}`, { cause: err }); // 重新抛出时保留原始错误
}
```

几个设计上的取舍：
- **整份配置校验失败就全部回退默认值。** 如果希望坏一个字段只影响这个字段，可以给每个字段加 `.catch(默认值)`。
- **`body` 如果来自请求，它是调用方可以随意构造的。** 所以 schema 里不要放 `role`、`isAdmin`、`tenantId` 这类字段。身份和权限要从认证信息或服务端数据里取。zod 默认会丢掉未声明的字段，下面有一条测试专门锁住这个行为。

import { describe, it, expect } from 'vitest';
import { getErrorMessage } from './errors';

describe('getErrorMessage', () => {
  it.each<[string, unknown, string]>([
    ['Error', new Error('boom'), 'boom'],
    ['Error subclass', new TypeError('bad'), 'bad'],
    ['string', 'plain', 'plain'],
    ['message-like object (cross-realm)', { message: 'from obj' }, 'from obj'],
    ['non-string message', { message: 42 }, '{"message":42}'],
    ['null', null, 'null'],
    ['undefined', undefined, 'undefined'],
    ['number', 404, '404'],
    ['bigint', 10n, '10'],
  ])('%s', (_, input, expected) => {
    expect(getErrorMessage(input)).toBe(expected);
  });

  it('circular object does not throw', () => {
    const o: Record<string, unknown> = {};
    o.self = o;
    expect(getErrorMessage(o)).toBe('[object Object]');
  });

  it('null-prototype circular object does not throw', () => {
    const o = Object.create(null) as Record<string, unknown>;
    o.self = o;
    expect(getErrorMessage(o)).toBe('Unknown error');

import { describe, it, expect, expectTypeOf } from 'vitest';
import { parseUser } from './user';
import { safeJsonParse, type Result } from './safe/json';

const valid = { id: 'u1', name: 'Ann', email: 'ann@example.com' };

describe('parseUser', () => {
  it('accepts a valid user', () => {
    expect(parseUser(JSON.stringify(valid))).toEqual({ ok: true, value: valid });
  });

  it('returns an error for malformed JSON instead of throwing', () => {
    const r = parseUser('{"id":');
    expect(r.ok).toBe(false);
    if (!r.ok) expect(r.error).toMatch(/^invalid JSON/);
  });

  it.each<[string, unknown, string]>([
    ['missing field', { id: 'u1', name: 'Ann' }, 'email'],
    ['wrong type', { ...valid, id: 123 }, 'id'],
    ['bad email', { ...valid, email: 'x' }, 'email'],
    ['negative age', { ...valid, age: -1 }, 'age'],
  ])('rejects %s', (_, input, path) => {
    const r = parseUser(JSON.stringify(input));
    expect(r.ok).toBe(false);
    if (!r.ok) expect(r.error).toContain(path);
  });

  it.each(['null', '[]', '"str"', '42'])('rejects non-object JSON %s', (text) => {
    expect(parseUser(text).ok).toBe(false);
  });

  it('strips forged privilege fields (isAdmin/role)', () => {
    const r = parseUser(JSON.stringify({ ...valid, isAdmin: true, role: 'admin' }));
    expect(r.ok).toBe(true);
    if (r.ok) {
      expect(r.value).toEqual(valid);
      expect(r.value).not.toHaveProperty('isAdmin');
    }
  });

  it('type: safeJsonParse never leaks any', () => {
    expectTypeOf(safeJsonParse('1')).toEqualTypeOf<Result<unknown>>();
  });
});

```ts
// src/config.test.ts
import { describe, it, expect, vi } from 'vitest';
import { loadConfig, defaultConfig, CONFIG_KEY } from './config';

const storageWith = (raw: string | null) => ({ getItem: vi.fn((_: string) => raw) });

describe('loadConfig', () => {
  it('reads the cfg key', () => {
    const s = storageWith(null);
    loadConfig(s);
    expect(s.getItem).toHaveBeenCalledWith(CONFIG_KEY);
  });

  it('returns defaults when key is missing', () => {
    expect(loadConfig(storageWith(null))).toEqual(defaultConfig());
  });

  it('loads valid config and fills missing fields with defaults', () => {
    expect(loadConfig(storageWith('{"theme":"dark"}'))).toEqual({ theme: 'dark', pageSize: 20, features: [] });
  });

  it('falls back on corrupted JSON and reports why', () => {
    const onInvalid = vi.fn();
    expect(loadConfig(storageWith('{oops'), onInvalid)).toEqual(defaultConfig());
    expect(onInvalid).toHaveBeenCalledWith(expect.stringMatching(/^invalid JSON/));
  });

  it.each(['{"pageSize":"20"}', '{"theme":"neon"}', '{"pageSize":100000}', '[]', 'null'])(
    'falls back on schema mismatch: %s',
    (raw) => {
      const onInvalid = vi.fn();
      expect(loadConfig(storageWith(raw), onInvalid)).toEqual(defaultConfig());
      expect(onInvalid).toHaveBeenCalledOnce();
    },
  );

  it('falls back when storage access throws', () => {
    const onInvalid = vi.fn();
    const s = { getItem: () => { throw new Error('SecurityError: denied'); } };
    expect(loadConfig(s, onInvalid)).toEqual(defaultConfig());
    expect(onInvalid).toHaveBeenCalledWith(expect.stringContaining('SecurityError'));
  });

  it('returns a fresh default object each call', () => {
    const a = loadConfig(storageWith(null));
    a.features.push('x');
    expect(loadConfig(storageWith(null)).features).toEqual([]);
  });
});
```

`loadConfig` 接收一个只需要 `getItem` 的存储对象作为参数，所以测试不依赖 jsdom，也能覆盖“访问存储直接抛错”这种在 jsdom 里很难模拟的情况。

1. 打开第 1 节的 lint 规则，先拿到一份完整的待改清单。
2. 每个 `JSON.parse(x) as T` 改成 `parseWith(Schema, x)`，并删掉对应的手写 interface。

## 思路

这三处的问题相同：`JSON.parse` 返回的是 `any`，`as X` 只让编译器不报错，运行时什么也不检查；`catch` 里的值也可能根本不是 `Error`。改法分三层：

1. **入口一律当 `unknown`**：外部输入（请求体、localStorage、`catch` 捕获的值）先当成 `unknown`。
2. **用 schema 在运行时校验，类型从 schema 推出来**：用 zod，`type User = z.infer<typeof UserSchema>`。校验和类型共用一份定义，不会各改各的。
3. **用编译选项和 lint 防止旧写法回来**：
   - `strict: true`（它包含 `useUnknownInCatchVariables`）
   - typescript-eslint 的 `no-unsafe-assignment` / `no-unsafe-member-access` / `no-unsafe-argument`
   - `consistent-type-assertions: ["error", { assertionStyle: "never" }]`（`as const` 仍然允许）
   - `use-unknown-in-catch-callback-variable`

   只加 ts-reset 不够：它让 `JSON.parse` 返回 `unknown`，但 `unknown as User` 照样能编译。真正挡住断言的是 `assertionStyle: "never"` 这条 lint。

**安全：** 请求体由调用方控制。像 `id`、`role`、`isAdmin` 这类身份和权限字段，要从登录态或服务端状态里取，不能信请求体。zod 的 `z.object` 默认会去掉未声明的字段，能挡住多余字段写入（mass assignment），下面有测试覆盖。

// src/safe/errors.ts
function describe(v: unknown): string {
  if (typeof v === 'string') return v;
  try { const s = JSON.stringify(v); if (s !== undefined) return s; } catch { /* 循环引用、BigInt */ }
  try { return String(v); } catch { return 'Unknown error'; } // Object.create(null) 等情况
}

export function toError(e: unknown): Error {
  if (e instanceof Error) return e;
  // 跨 realm（iframe、worker）抛出的 Error 不满足 instanceof，按形状识别
  if (typeof e === 'object' && e !== null && 'message' in e && typeof e.message === 'string') {
    return new Error(e.message, { cause: e });
  }
  return new Error(describe(e), { cause: e });
}

export const getErrorMessage = (e: unknown): string => toError(e).message;

```ts
// src/safe/json.ts
import type { z } from 'zod';
import { toError } from './errors';

export type Result<T, E = Error> = { ok: true; value: T } | { ok: false; error: E };
export type Issue = { path: string; message: string };

export class ParseError extends Error {
  readonly kind: 'syntax' | 'schema';
  readonly issues: readonly Issue[];
  constructor(kind: 'syntax' | 'schema', message: string, issues: readonly Issue[] = [], options?: ErrorOptions) {
    super(message, options);
    this.name = 'ParseError';
    this.kind = kind;
    this.issues = issues;
  }
}

export function parseJson(text: string): Result<unknown, ParseError> {
  try {
    const value: unknown = JSON.parse(text); // 注解成 unknown 就够了，不需要 as
    return { ok: true, value };
  } catch (e) {
    return { ok: false, error: new ParseError('syntax', toError(e).message, [], { cause: e }) };
  }
}

export function parseWith<T>(schema: z.ZodType<T>, text: string): Result<T, ParseError> {
  const json = parseJson(text);
  if (!json.ok) return json;
  const r = schema.safeParse(json.value);
  if (r.success) return { ok: true, value: r.data };
  const issues = r.error.issues.map((i) => ({ path: i.path.map(String).join('.'), message: i.message }));
  return { ok: false, error: new ParseError('schema', 'Invalid shape', issues) };
}
```

```ts
// src/config.ts
import { z } from 'zod';
import { parseWith, type Result } from './safe/json';
import { toError } from './safe/errors';

const KEY = 'cfg';
export const ConfigSchema = z.object({
  version: z.literal(1),
  theme: z.enum(['light', 'dark']).default('light'),
  pageSize: z.number().int().min(1).max(200).default(20),
});
export type Config = z.infer<typeof ConfigSchema>;
export const DEFAULT_CONFIG: Config = ConfigSchema.parse({ version: 1 });

type ReadStore = Pick<Storage, 'getItem'>;
type WriteStore = Pick<Storage, 'setItem'>;

// Safari 隐私模式、禁用 cookie 时，光是访问 localStorage 就可能抛 SecurityError
export function safeLocalStorage(): Storage | undefined {
  try { return globalThis.localStorage; } catch { return undefined; }
}

export function loadConfig(
  store: ReadStore | undefined = safeLocalStorage(),
  onInvalid: (err: Error) => void = () => {},
): Config {
  let raw: string | null;
  try { raw = store?.getItem(KEY) ?? null; } catch (e) { onInvalid(toError(e)); return DEFAULT_CONFIG; }
  if (raw === null) return DEFAULT_CONFIG;
  const r = parseWith(ConfigSchema, raw);
  if (!r.ok) { onInvalid(r.error); return DEFAULT_CONFIG; } // 旧版本或被篡改都回退到默认值
  return r.value;
}

export function saveConfig(cfg: Config, store: WriteStore | undefined = safeLocalStorage()): Result<void> {
  if (!store) return { ok: false, error: new Error('storage unavailable') };
  try { store.setItem(KEY, JSON.stringify(ConfigSchema.parse(cfg))); return { ok: true, value: undefined }; }
  catch (e) { return { ok: false, error: toError(e) }; } // 例如 QuotaExceededError
}
```

`catch` 的改法：

// src/user.test.ts
import { describe, it, expect, expectTypeOf } from 'vitest';
import { parseWith } from './safe/json';
import { UserSchema, type User } from './user';

const ok = JSON.stringify({ id: 'u1', name: 'A', email: 'a@example.com' });

describe('parseWith(UserSchema)', () => {
  it('合法输入', () => {
    const r = parseWith(UserSchema, ok);
    expect(r.ok).toBe(true);
    if (r.ok) expectTypeOf(r.value).toEqualTypeOf<User>();
  });
  it('JSON 语法错误 → syntax', () => {
    const r = parseWith(UserSchema, '{bad');
    expect(!r.ok && r.error.kind).toBe('syntax');
  });
  it.each(['null', '[]', '"str"', '{}'])('结构不对 %s → schema', (t) => {
    const r = parseWith(UserSchema, t);
    expect(!r.ok && r.error.kind).toBe('schema');
  });
  it('字段类型错误时报出路径', () => {
    const r = parseWith(UserSchema, JSON.stringify({ id: 'u1', name: 'A', email: 123 }));
    expect(r.ok ? [] : r.error.issues.map((i) => i.path)).toContain('email');
  });
  it('越权字段被剥离（负向用例）', () => {
    const r = parseWith(UserSchema, JSON.stringify({ id: 'u1', name: 'A', email: 'a@example.com', isAdmin: true, role: 'admin' }));
    expect(r.ok && r.value).not.toHaveProperty('isAdmin');
    expect(r.ok && r.value).not.toHaveProperty('role');
  });
  it('__proto__ 不会污染原型', () => {
    parseWith(UserSchema, '{"id":"u1","name":"A","email":"a@example.com","__proto__":{"polluted":1}}');
    expect(({} as Record<string, unknown>).polluted).toBeUndefined();
  });
});

// src/config.test.ts
import { describe, it, expect, vi } from 'vitest';
import { loadConfig, saveConfig, DEFAULT_CONFIG } from './config';

const mem = (init: Record<string, string> = {}) => {
  const m = new Map(Object.entries(init));
  return { getItem: (k: string) => m.get(k) ?? null, setItem: (k: string, v: string) => void m.set(k, v), m };
};

describe('loadConfig', () => {
  it('没有存储值 → 默认值', () => expect(loadConfig(mem())).toEqual(DEFAULT_CONFIG));
  it('没有 storage → 默认值', () => expect(loadConfig(undefined)).toEqual(DEFAULT_CONFIG));
  it('只存了部分字段 → 补齐默认值', () =>
    expect(loadConfig(mem({ cfg: '{"version":1,"theme":"dark"}' }))).toEqual({ version: 1, theme: 'dark', pageSize: 20 }));
  it.each([
    ['损坏的 JSON', '{oops'],
    ['类型错误', '{"version":1,"pageSize":"20"}'],
    ['越界', '{"version":1,"pageSize":99999}'],
    ['旧版本', '{"version":0}'],
    ['非对象', '"dark"'],
  ])('%s → 回退默认值并回调 onInvalid', (_, raw) => {
    const onInvalid = vi.fn();
    expect(loadConfig(mem({ cfg: raw }), onInvalid)).toEqual(DEFAULT_CONFIG);
    expect(onInvalid).toHaveBeenCalledOnce();
  });
  it('getItem 抛 SecurityError → 默认值', () => {
    const onInvalid = vi.fn();
    const store = { getItem: () => { throw new DOMException('denied', 'SecurityError'); } };
    expect(loadConfig(store, onInvalid)).toEqual(DEFAULT_CONFIG);
    expect(onInvalid).toHaveBeenCalledOnce();
  });
});

describe('saveConfig', () => {
  it('写入后能读回同样的值', () => {
    const s = mem();
    const cfg = { version: 1 as const, theme: 'dark' as const, pageSize: 50 };
    expect(saveConfig(cfg, s).ok).toBe(true);
    expect(loadConfig(s)).toEqual(cfg);
  });
  it('配额满时返回错误，不抛异常', () => {
    const s = { setItem: () => { throw new DOMException('full', 'QuotaExceededError'); } };
    const r = saveConfig(DEFAULT_CONFIG, s);
    expect(!r.ok && r.error.name).toBe('QuotaExceededError');
  });
  it('编译期拒绝非法配置', () => {
    // @ts-expect-error theme 只能是 'light' 或 'dark'
    saveConfig({ version: 1, theme: 'blue', pageSize: 20 }, mem());
  });

## 做法

1. **外部数据一律先当 `unknown`，在入口处检查一次结构，得到准确类型**：`JSON.parse` 的结果、`localStorage` 读出的值、`catch` 里的变量都属于这类。后面的业务代码只拿检查过的值。
2. **用运行时 schema 代替 `as`**：`as User` 只是让编译器不再检查，运行时什么也没验证。下面用 zod 举例；如果项目里已经有别的 schema 库（valibot、io-ts 等），就用已有的，不要再加一个。类型从 schema 推出来（`z.infer`），只维护一份定义。
3. **失败用返回值表达，不抛异常、不做强转**：返回一个带 `ok` 标记的结果类型，调用方必须先判断成功才能读到值。
4. **加门禁防止以后再出现**：`strict`（其中包含 `useUnknownInCatchVariables`）+ 带类型信息的 lint 规则 + 禁止 `JSON.parse(...) as T` 的语法规则。

> 安全提示：如果 `body` 是客户端提交的，结构合法不等于内容可信。`id`、`role`、租户这类身份和权限字段要从登录态或服务端数据取，不能用请求体里的值。下面的 `UserSchema` 用 `.strict()`，多出来的字段（比如伪造的 `role`）会被直接拒绝，测试里有这个用例。

```ts
export type Result<T, E> = { ok: true; value: T } | { ok: false; error: E };

export type BoundaryError =
  | { kind: 'invalid_json'; message: string }
  | { kind: 'invalid_shape'; issues: { path: string; message: string }[] };
```

**`src/boundary/json.ts`**
```ts
import type { z } from 'zod';
import type { BoundaryError, Result } from './result';

/** JSON.parse 返回的是 any，这里在入口处收窄成 unknown。 */
export function safeJsonParse(text: string): Result<unknown, BoundaryError> {
  try {
    const value: unknown = JSON.parse(text);
    return { ok: true, value };
  } catch (err) {
    return { ok: false, error: { kind: 'invalid_json', message: errorMessage(err) } };
  }
}

export function parseWith<T>(schema: z.ZodType<T>, input: unknown): Result<T, BoundaryError> {
  const r = schema.safeParse(input);
  if (r.success) return { ok: true, value: r.data };
  return {
    ok: false,
    error: {
      kind: 'invalid_shape',
      issues: r.error.issues.map((i) => ({ path: i.path.map(String).join('.'), message: i.message })),
    },
  };
}

export function parseJsonAs<T>(schema: z.ZodType<T>, text: string): Result<T, BoundaryError> {
  const json = safeJsonParse(text);
  return json.ok ? parseWith(schema, json.value) : json;
}

import { errorMessage } from './errors';
```

**`src/boundary/errors.ts`**：替代 `err.message`
```ts
function safeString(value: unknown): string {
  try {
    return JSON.stringify(value) ?? String(value);
  } catch {
    try { return String(value); } catch { return '[unrepresentable thrown value]'; }
  }
}

/** 把任意被 throw 的值转成 Error，并把原值保存在 cause 里。 */
export function toError(value: unknown): Error {
  if (value instanceof Error) return value;
  if (typeof value === 'string') return new Error(value);
  // 来自其他 realm（iframe、vm）的 Error 用 instanceof 判断不出来，按结构识别
  if (typeof value === 'object' && value !== null && 'message' in value && typeof value.message === 'string') {
    return new Error(value.message, { cause: value });
  }
  return new Error(safeString(value), { cause: value });
}

export const errorMessage = (value: unknown): string => toError(value).message;
```

**`src/config.ts`**：替代 `localStorage.getItem('cfg') as Config`
```ts
import { z } from 'zod';
import { parseJsonAs } from './boundary/json';
import type { BoundaryError } from './boundary/result';

export const ConfigSchema = z.object({
  theme: z.enum(['light', 'dark']).default('light'),
  pageSize: z.number().int().min(10).max(100).default(20),
  features: z.object({ beta: z.boolean().default(false) }).default({ beta: false }),
}); // 配置用默认的剥离模式：旧版本留下的字段直接丢掉

export type Config = z.infer<typeof ConfigSchema>;
export const defaultConfig = (): Config => ConfigSchema.parse({});

export type ConfigLoad =
  | { source: 'stored'; config: Config }
  | { source: 'default'; config: Config; reason: 'unavailable' | 'missing' | BoundaryError['kind'] };

type ReadableStorage = Pick<Storage, 'getItem'>;

function browserStorage(): ReadableStorage | null {
  try { return globalThis.localStorage ?? null; } catch { return null; } // 隐私模式或禁用存储时访问会抛 SecurityError
}

export function loadConfig(storage: ReadableStorage | null = browserStorage()): ConfigLoad {
  let raw: string | null;
  try {
    raw = storage ? storage.getItem('cfg') : null;
    if (!storage) return { source: 'default', config: defaultConfig(), reason: 'unavailable' };
  } catch {
    return { source: 'default', config: defaultConfig(), reason: 'unavailable' };
  }
  if (raw === null) return { source: 'default', config: defaultConfig(), reason: 'missing' };

  const r = parseJsonAs(ConfigSchema, raw);
  return r.ok
    ? { source: 'stored', config: r.value }
    : { source: 'default', config: defaultConfig(), reason: r.error.kind };
}
```

## 防止回退的门禁

**tsconfig**
```jsonc
{ "compilerOptions": { "strict": true, "noUncheckedIndexedAccess": true, "lib": ["ES2022", "DOM"] } }
```
如果暂时开不了完整的 `strict`，至少打开 `"useUnknownInCatchVariables": true`。`Error` 的 `cause` 参数需要 lib 至少到 ES2022。

**ESLint**（typescript-eslint，需要带类型信息的配置）
```js
rules: {
  '@typescript-eslint/no-explicit-any': 'error',
  '@typescript-eslint/no-unsafe-assignment': 'error',   // 拦截 const u: User = JSON.parse(x)
  '@typescript-eslint/no-unsafe-member-access': 'error',
  '@typescript-eslint/no-unsafe-argument': 'error',
  '@typescript-eslint/no-unsafe-return': 'error',
  '@typescript-eslint/use-unknown-in-catch-callback-variable': 'error', // 拦截 .catch((e: any) => …)
  'no-restricted-syntax': ['error',
    { selector: 'TSAsExpression > CallExpression[callee.object.name="JSON"][callee.property.name="parse"]',
      message: '不要对 JSON.parse 的结果用 as，改用 parseJsonAs(schema, text)' },
    { selector: 'TSTypeAssertion > CallExpression[callee.object.name="JSON"][callee.property.name="parse"]',
      message: '同上' },
    { selector: 'TSAsExpression > TSAsExpression > TSUnknownKeyword',
      message: '`as unknown as T` 会绕过所有检查' },
  ],
}
```
`no-unsafe-*` 这组规则拦不住 `JSON.parse(x) as User`，因为一加 `as` 值就不再是 `any`。只改 TS 类型也没用：即使让 `JSON.parse` 返回 `unknown`，`unknown as User` 照样能编译。所以要靠这条 `no-restricted-syntax` 规则来禁。

```ts
import { describe, expect, it } from 'vitest';
import { parseUser } from './user';

const ok = { id: 'u1', name: 'Ann', email: 'a@example.com' };

describe('parseUser', () => {
  it('合法输入', () => {
    expect(parseUser(JSON.stringify(ok))).toEqual({ ok: true, value: ok });
  });
  it('非法 JSON → invalid_json', () => {
    const r = parseUser('{not json');
    expect(r.ok || r.error.kind).toBe('invalid_json');
  });
  it.each([
    ['字段类型错', { ...ok, email: 123 }, 'email'],
    ['缺字段', { name: 'Ann', email: 'a@example.com' }, 'id'],
    ['age 非整数', { ...ok, age: 1.5 }, 'age'],
  ])('%s → invalid_shape 且指出路径', (_n, input, path) => {
    const r = parseUser(JSON.stringify(input));
    expect(r.ok).toBe(false);
    if (!r.ok && r.error.kind === 'invalid_shape') expect(r.error.issues.map((i) => i.path)).toContain(path);
    else throw new Error('expected invalid_shape');
  });
  it('伪造多余字段（role）被拒绝', () => {
    expect(parseUser(JSON.stringify({ ...ok, role: 'admin' })).ok).toBe(false);
  });
  it('__proto__ 键被拒绝', () => {
    expect(parseUser(`{"id":"u1","name":"A","email":"a@example.com","__proto__":{"isAdmin":true}}`).ok).toBe(false);
  });
  it.each(['null', '[]', '"str"', '42'])('顶层不是对象：%s', (s) => {
    expect(parseUser(s).ok).toBe(false);
  });
});
```

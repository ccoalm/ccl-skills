## 思路

外部数据（请求体、localStorage、catch 到的东西）进来时一律当 `unknown`，在这个入口校验一次，校验通过才拿到具体类型。同时用编译器和 lint 禁止再写 `as`，防止以后有人又绕过去。

| 问题 | 为什么不安全 | 改法 |
|---|---|---|
| `JSON.parse(body) as User` | `JSON.parse` 返回 `any`，`as` 只是告诉编译器"相信我"，运行时什么都不检查 | 用 zod schema 校验，再从 schema 推出类型 |
| `localStorage.getItem('cfg')` 直接断言 | 可能是 `null`、JSON 损坏、旧版本的格式，或者 `getItem` 本身抛 `SecurityError`（如隐私模式） | 每种失败都回退到默认配置 |
| catch 里直接用 `err.message` | 被抛出的可以是任何值：字符串、`undefined`、跨 realm 的 Error | 打开 `useUnknownInCatchVariables`，统一用 `toError()` 转换 |

## 2. 实现

```ts
// src/safe/errors.ts
export function toError(e: unknown): Error {
  if (e instanceof Error) return e;
  // 跨 iframe/realm 的 Error 用 instanceof 判断会失败，这里退而检查 message 字段
  if (typeof e === 'object' && e !== null && 'message' in e && typeof e.message === 'string') {
    return new Error(e.message, { cause: e });
  }
  if (typeof e === 'string') return new Error(e);
  let text: string;
  try {
    text = JSON.stringify(e) ?? String(e);   // stringify(undefined) 返回 undefined
  } catch {
    text = String(e);                        // 循环引用或 BigInt 会让 stringify 抛错
  }
  return new Error(text, { cause: e });
}

export const errorMessage = (e: unknown): string => toError(e).message;
```

```ts
// src/safe/json.ts
import { z } from 'zod'; // 按 zod v4 写
import { toError } from './errors';

export type Result<T, E = Error> = { ok: true; value: T } | { ok: false; error: E };

export function parseJson(text: string): Result<unknown> {
  try {
    const value: unknown = JSON.parse(text);   // 用类型标注，不用 as
    return { ok: true, value };
  } catch (e) {
    return { ok: false, error: toError(e) };
  }
}

export function parseJsonAs<S extends z.ZodType>(schema: S, text: string): Result<z.output<S>> {
  const raw = parseJson(text);
  if (!raw.ok) return raw;
  const r = schema.safeParse(raw.value);
  return r.success
    ? { ok: true, value: r.data }
    : { ok: false, error: new Error(`Invalid shape:\n${z.prettifyError(r.error)}`, { cause: r.error }) };
}
```

> ⚠️ **安全提醒**：校验只能说明"格式对"，不能说明"内容可信"。如果请求体里的 `User` 带了 `id` 或 `role`，鉴权仍然要用认证会话里的身份，不能用请求体里的。zod 默认会丢掉 schema 里没声明的字段（下面的测试覆盖了这一点）。

```ts
// src/config/storage.ts
import { z } from 'zod';
import { parseJsonAs } from '../safe/json';

export const ConfigSchema = z.object({
  version: z.literal(1),
  theme: z.enum(['light', 'dark']),
  pageSize: z.number().int().min(1).max(200),
});
export type Config = z.infer<typeof ConfigSchema>;
export const DEFAULT_CONFIG: Config = { version: 1, theme: 'light', pageSize: 20 };

type ReadableStorage = Pick<Storage, 'getItem'>;

function defaultStorage(): ReadableStorage | undefined {
  try { return globalThis.localStorage; } catch { return undefined; } // 有些环境里光是访问就会抛错
}

// 传入 storage 参数是为了方便测试
export function readConfig(storage: ReadableStorage | undefined = defaultStorage()): Config {
  if (!storage) return DEFAULT_CONFIG;
  let text: string | null;
  try { text = storage.getItem('cfg'); } catch { return DEFAULT_CONFIG; }
  if (text === null) return DEFAULT_CONFIG;
  const r = parseJsonAs(ConfigSchema, text);
  return r.ok ? r.value : DEFAULT_CONFIG; // 不主动删除：可能是新版本写入的数据，用户回退版本后不该丢
}
```

如果希望某个字段坏了不影响其它字段，可以给单个字段加 `.catch(默认值)`，例如 `theme: z.enum([...]).catch('light')`。

```ts
// src/safe/json.test.ts
import { describe, it, expect, expectTypeOf } from 'vitest';
import { parseJsonAs } from './json';
import { UserSchema, type User } from '../models/user';

describe('parseJsonAs(UserSchema)', () => {
  it('accepts a valid user and infers User', () => {
    const r = parseJsonAs(UserSchema, '{"id":"u1","name":"Ann","age":3}');
    expect(r).toEqual({ ok: true, value: { id: 'u1', name: 'Ann', age: 3 } });
    if (r.ok) expectTypeOf(r.value).toEqualTypeOf<User>();
  });

  it('returns an error (no throw) for malformed JSON', () => {
    const r = parseJsonAs(UserSchema, '{bad');
    expect(r.ok).toBe(false);
    if (!r.ok) expect(r.error).toBeInstanceOf(Error);
  });

  it.each(['null', '[]', '"str"', '{"id":1,"name":"a"}', '{"id":"1"}', '{"id":"1","name":""}', '{"id":"1","name":"a","age":-1}'])(
    'rejects wrong shape %s', (text) => {
      expect(parseJsonAs(UserSchema, text).ok).toBe(false);
    });

  it('strips undeclared fields such as role', () => {
    const r = parseJsonAs(UserSchema, '{"id":"1","name":"a","role":"admin"}');
    expect(r.ok && r.value).not.toHaveProperty('role');
  });
});
```

```ts
// src/config/storage.test.ts
import { describe, it, expect } from 'vitest';
import { readConfig, DEFAULT_CONFIG } from './storage';

const store = (v: string | null) => ({ getItem: () => v });

describe('readConfig', () => {
  it('returns stored config when valid', () => {
    const cfg = { version: 1, theme: 'dark', pageSize: 50 };
    expect(readConfig(store(JSON.stringify(cfg)))).toEqual(cfg);
  });
  it.each([
    ['missing key', null],
    ['corrupt JSON', '{"theme":'],
    ['wrong shape', '{"version":1,"theme":"pink","pageSize":20}'],
    ['future version', '{"version":2,"theme":"dark","pageSize":20}'],
    ['not an object', '42'],
  ])('falls back to default on %s', (_, v) => {
    expect(readConfig(store(v))).toEqual(DEFAULT_CONFIG);
  });
  it('falls back when getItem throws (e.g. SecurityError)', () => {
    const throwing = { getItem: () => { throw new DOMException('denied', 'SecurityError'); } };
    expect(readConfig(throwing)).toEqual(DEFAULT_CONFIG);
  });
  it('falls back when storage is unavailable', () => {
    expect(readConfig(undefined)).toEqual(DEFAULT_CONFIG);
  });
});
```

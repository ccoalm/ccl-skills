## 思路

`as User` 只是告诉编译器“相信我”，运行时不做任何检查。改法分三步：

1. **外部数据先当 `unknown`，再用运行时校验收窄类型。** 用 zod 定义 schema，`type` 从 schema 推出来，校验和类型只有一份来源。
2. **出错时给明确结果。** 失败返回 `Result`，或者退回默认值，不让脏数据带着错误的类型流进业务代码。
3. **用编译选项和 lint 堵住回退。** 只改现有代码，以后还会有人写新的 `as User`，要靠规则拦。

## 1. 安全解析 JSON

```ts
// src/lib/result.ts
export type Result<T, E = Error> =
  | { ok: true; value: T }
  | { ok: false; error: E };
```

```ts
// src/lib/safe-json.ts
import { z } from 'zod';
import type { Result } from './result';

export class ParseError extends Error {
  constructor(
    message: string,
    readonly issues: z.ZodError['issues'] = [],
    options?: ErrorOptions,
  ) {
    super(message, options);
    this.name = 'ParseError';
  }
}

export function parseJson(text: string): Result<unknown, ParseError> {
  try {
    const value: unknown = JSON.parse(text); // 显式标注 unknown，不用 as
    return { ok: true, value };
  } catch (err) {
    return { ok: false, error: new ParseError('Invalid JSON', [], { cause: err }) };
  }
}

export function parseJsonAs<T>(schema: z.ZodType<T>, text: string): Result<T, ParseError> {
  const raw = parseJson(text);
  if (!raw.ok) return raw;
  const r = schema.safeParse(raw.value);
  return r.success
    ? { ok: true, value: r.data }
    : { ok: false, error: new ParseError('Schema mismatch', r.error.issues) };
}
```

```ts
// src/models/user.ts
import { z } from 'zod';

export const UserSchema = z.object({
  id: z.string().min(1),
  name: z.string().min(1).max(100),
  email: z.string().email(),
  age: z.number().int().nonnegative().optional(),
});
export type User = z.infer<typeof UserSchema>;

// 客户端传进来的入参单独定义：不含 id、role 这类字段
export const CreateUserInputSchema = UserSchema.omit({ id: true });
export type CreateUserInput = z.infer<typeof CreateUserInputSchema>;
```

调用处：

```ts
// 之前：const user = JSON.parse(body) as User
const parsed = parseJsonAs(CreateUserInputSchema, body);
if (!parsed.ok) return reply(400, { error: 'invalid body', issues: parsed.error.issues });
const input = parsed.value; // 类型是 CreateUserInput，运行时也已校验过
```

**安全注意：** `body` 由调用方完全控制。zod 的 `z.object` 默认会丢掉没有声明的字段，所以 `{"role":"admin"}` 这类多塞的字段进不来，`__proto__` 也一样。`id`、`role`、租户这类身份字段应该从认证后的会话里取，不能从请求体里读，所以入参 schema 里干脆不写这些字段。如果用 Express，`req.body` 是 `any`，同样要过一遍 `CreateUserInputSchema.safeParse(req.body)`。

## 2. 读 localStorage 配置

存进去的配置可能是旧版本、被手动改过、已经损坏，或者当前环境根本不能访问 localStorage（Safari 隐私模式、禁用 cookie 时访问会抛 `SecurityError`）。

```ts
// src/config/config.ts
import { z } from 'zod';
import { parseJsonAs } from '../lib/safe-json';
import { toError } from '../lib/errors';

const KEY = 'cfg';

// 每个字段用 .catch：某个字段坏了只回退这一个，不会把整份配置清掉
const ConfigSchema = z.object({
  version: z.literal(1),
  theme: z.enum(['light', 'dark']).catch('light'),
  pageSize: z.number().int().min(10).max(100).catch(20),
});
export type Config = z.infer<typeof ConfigSchema>;

export const defaultConfig = (): Config => ({ version: 1, theme: 'light', pageSize: 20 });

type StorageLike = Pick<Storage, 'getItem' | 'setItem'>;

function browserStorage(): StorageLike | undefined {
  try {
    return typeof localStorage === 'undefined' ? undefined : localStorage;
  } catch {
    return undefined; // 访问本身就可能抛错
  }
}

export function loadConfig(
  storage: StorageLike | undefined = browserStorage(),
  onError: (e: Error) => void = () => {},
): Config {
  let text: string | null = null;
  try {
    text = storage?.getItem(KEY) ?? null;
  } catch (err) {
    onError(toError(err));
    return defaultConfig();
  }
  if (text === null) return defaultConfig();

  const r = parseJsonAs(ConfigSchema, text);
  if (r.ok) return r.value;
  onError(r.error); // 不是对象、版本不对、JSON 坏了，都走这里
  return defaultConfig();
}

export function saveConfig(cfg: Config, storage = browserStorage()): Result<void> {
  try {
    storage?.setItem(KEY, JSON.stringify(ConfigSchema.parse(cfg)));
    return { ok: true, value: undefined };
  } catch (err) {
    return { ok: false, error: toError(err) }; // 例如 QuotaExceededError
  }
}
```

（`saveConfig` 需要 `import type { Result } from '../lib/result'`。）以后升级配置格式时，把 `version` 改成 `z.union` 加一个迁移函数即可。

## 3. 安全处理 catch 里的错误

```ts
// src/lib/errors.ts
export function toError(value: unknown): Error {
  if (value instanceof Error) return value;
  if (typeof value === 'string') return new Error(value);
  // 来自 iframe 等其他执行环境的 Error 过不了 instanceof，这里按结构判断
  if (typeof value === 'object' && value !== null && 'message' in value
      && typeof value.message === 'string') {
    return new Error(value.message, { cause: value });
  }
  let text: string;
  try {
    text = JSON.stringify(value) ?? String(value);
  } catch {
    // 循环引用、BigInt、无原型对象都会落到这里
    text = Object.prototype.toString.call(value);
  }
  return new Error(text, { cause: value });
}

export const getErrorMessage = (e: unknown): string => toError(e).message;
```

## 4. 用编译器和 lint 防止回退

`tsconfig.json`：

```jsonc
{ "compilerOptions": { "strict": true /* 包含 useUnknownInCatchVariables */ } }
```

`eslint.config.js`（使用 typescript-eslint 的 `recommendedTypeChecked`）：

```js
rules: {
  // unknown 和 any 都能直接 as 成任意类型，编译器不会报错，只有这条规则能拦住
  '@typescript-eslint/consistent-type-assertions': ['error', { assertionStyle: 'never' }], // as const 仍然允许
  '@typescript-eslint/use-unknown-in-catch-callback-variable': 'error', // 覆盖 .catch(err => ...)
  '@typescript-eslint/no-unsafe-assignment': 'error',
  '@typescript-eslint/no-unsafe-member-access': 'error',
  'no-restricted-properties': ['error',
    { object: 'JSON', property: 'parse', message: '请用 parseJsonAs(schema, text)' },
    { object: 'localStorage', property: 'getItem', message: '请走对应的 load* 函数' },
  ],
},
// 再加一个 override，让 src/lib/safe-json.ts 和 src/config/*.ts 可以使用这两个 API
```

如果全量开 `assertionStyle: 'never'` 一开始报错太多，可以先改成 `warn`，按目录逐步收紧。另外可以装 `@total-typescript/ts-reset`，它会把 `JSON.parse` 和 `fetch().json()` 的返回值从 `any` 改成 `unknown`。

**迁移顺序：**
1. 先打开上面的 lint 规则，拿到完整的待改清单，这比 grep 更全。
2. 按三类分别替换。
3. 每改完一类就跑一次 `tsc --noEmit` 和测试。

import { describe, it, expect, expectTypeOf } from 'vitest';
import { parseJsonAs } from './safe-json';
import { UserSchema, CreateUserInputSchema, type User } from '../models/user';

const valid = { id: 'u1', name: 'Ann', email: 'ann@example.com' };

describe('parseJsonAs', () => {
  it('合法输入 → ok 且类型为 User', () => {
    const r = parseJsonAs(UserSchema, JSON.stringify(valid));
    expect(r).toEqual({ ok: true, value: valid });
    if (r.ok) expectTypeOf(r.value).toEqualTypeOf<User>();
  });

  it.each([
    ['坏 JSON', '{"id":'],
    ['空串', ''],
    ['null', 'null'],
    ['数组', '[]'],
    ['缺字段', JSON.stringify({ id: 'u1' })],
    ['类型错', JSON.stringify({ ...valid, age: '18' })],
    ['邮箱非法', JSON.stringify({ ...valid, email: 'x' })],

    const r = parseJsonAs(UserSchema, text);
    expect(r.ok).toBe(false);
  });

  it('坏 JSON 的错误保留原始 cause', () => {
    const r = parseJsonAs(UserSchema, '{');
    expect(!r.ok && r.error.cause).toBeInstanceOf(SyntaxError);
  });

  it('伪造字段被剥离：role / id / __proto__ 不进入结果', () => {

    expect(r.ok).toBe(true);
    if (r.ok) {
      expect(r.value).toEqual({ name: 'Ann', email: 'a@example.org' });
      expect(({} as Record<string, unknown>).isAdmin).toBeUndefined(); // 没有原型污染
    }
  });
});

import { describe, it, expect, vi } from 'vitest';
import { loadConfig, saveConfig, defaultConfig } from './config';

const mem = (init?: string) => {
  const m = new Map<string, string>(init === undefined ? [] : [['cfg', init]]);
  return {
    getItem: (k: string) => m.get(k) ?? null,
    setItem: (k: string, v: string) => void m.set(k, v),
  };
};

describe('loadConfig', () => {
  it('没有存储过 → 默认值', () => expect(loadConfig(mem())).toEqual(defaultConfig()));
  it('没有 storage（SSR / Node）→ 默认值', () => expect(loadConfig(undefined)).toEqual(defaultConfig()));

  it('合法配置原样读出', () => {
    const cfg = { version: 1, theme: 'dark', pageSize: 50 };
    expect(loadConfig(mem(JSON.stringify(cfg)))).toEqual(cfg);
  });

  it('单个字段坏了只回退该字段', () => {
    const s = mem(JSON.stringify({ version: 1, theme: 'neon', pageSize: 50 }));
    expect(loadConfig(s)).toEqual({ version: 1, theme: 'light', pageSize: 50 });
  });

  it.each(['{broken', '"str"', 'null', '[]', JSON.stringify({ version: 2 })])(
    '损坏或版本不对（%s）→ 默认值，并上报错误',
    (raw) => {
      const onError = vi.fn();
      expect(loadConfig(mem(raw), onError)).toEqual(defaultConfig());
      expect(onError).toHaveBeenCalledOnce();
    },
  );

  it('getItem 抛 SecurityError → 默认值，不崩溃', () => {
    const onError = vi.fn();
    const s = { getItem: () => { throw new DOMException('denied', 'SecurityError'); }, setItem: () => {} };
    expect(loadConfig(s, onError)).toEqual(defaultConfig());
    expect(onError.mock.calls[0][0]).toBeInstanceOf(Error);

  it('返回的默认值互相独立，修改一份不影响下一次', () => {

    expect(loadConfig(mem()).pageSize).toBe(20);

## 思路

`JSON.parse` 的结果、`localStorage` 读出的值、`catch` 拿到的错误，都来自程序外部。TypeScript 的类型在运行时不存在，`as User` 只是让编译器不再检查，并没有检查数据本身。正确做法是：这类值一进来先当作 `unknown`，在入口处用运行时 schema 校验一次，得到准确的类型，后面的业务代码就不用再判断了。

- **解析失败**：返回一个带类型的结果，调用方必须先判断成功还是失败，不能靠断言跳过。
- **静态类型**：从 schema 推导出来（`z.output`），不再单独写一份 interface。
- **防止回退**：用 `strict` 加 typed lint 卡住，以后再出现 `as` 断言或直接用 `err.message` 都会报错。

下面用 zod 4 写。如果仓库里已经有别的 schema 库（valibot、arktype 等），就用现有的，不要再加一个。

```ts
export type Result<T, E> = { ok: true; value: T } | { ok: false; error: E };

export type Issue = { path: string; message: string };
export type BoundaryError =
  | { kind: 'invalid-json'; message: string }
  | { kind: 'invalid-shape'; issues: Issue[] };
```

```ts
// src/boundary/error.ts
/** catch 变量是 unknown：可能是 Error、字符串、普通对象、null，或来自 iframe 的 Error（instanceof 判断会失败） */
export function getErrorMessage(err: unknown): string {
  if (err instanceof Error) return err.message;
  if (typeof err === 'string') return err;
  if (typeof err === 'object' && err !== null && 'message' in err && typeof err.message === 'string') {
    return err.message;
  }
  try {
    return JSON.stringify(err) ?? String(err); // undefined 或函数会让 JSON.stringify 返回 undefined
  } catch { /* 循环引用、BigInt 会抛错 */ }
  try {
    return String(err); // Object.create(null) 或 toString 会抛错的对象，这里也会抛
  } catch {
    return 'Unknown error';
  }
}

/** 需要重新抛出时用：保留原始错误作为 cause（需要 lib >= es2022） */
export function toError(err: unknown): Error {
  return err instanceof Error ? err : new Error(getErrorMessage(err), { cause: err });
}
```

```ts
// src/boundary/parse.ts
import { z } from 'zod';
import type { BoundaryError, Issue, Result } from './result';
import { getErrorMessage } from './error';

const toIssues = (e: z.ZodError): Issue[] =>
  e.issues.map((i) => ({ path: i.path.map(String).join('.'), message: i.message })); // path 里可能有 symbol，直接 join 会抛错

export function parseWith<S extends z.ZodType>(schema: S, text: string): Result<z.output<S>, BoundaryError> {
  let data: unknown;
  try {
    data = JSON.parse(text);
  } catch (err) {
    return { ok: false, error: { kind: 'invalid-json', message: getErrorMessage(err) } };
  }
  return validateWith(schema, data);
}

/** 用于框架已经解析好的 body（如 Express 的 req.body，它的类型是 any） */
export function validateWith<S extends z.ZodType>(schema: S, data: unknown): Result<z.output<S>, BoundaryError> {
  const r = schema.safeParse(data);
  return r.success
    ? { ok: true, value: r.data }
    : { ok: false, error: { kind: 'invalid-shape', issues: toIssues(r.error) } };
}
```

## 2. 替换 `JSON.parse(body) as User`

```ts
// src/user/user.ts
import { z } from 'zod';
import { parseWith } from '../boundary/parse';

export const UserSchema = z.strictObject({ // 出现未知字段就拒绝
  id: z.string().min(1),
  name: z.string().min(1).max(100),
  email: z.email(),
  age: z.number().int().nonnegative().optional(),
});
export type User = z.output<typeof UserSchema>; // 删掉原来手写的 interface User

export const parseUser = (body: string) => parseWith(UserSchema, body);
```

调用处这样写：

```ts
const parsed = parseUser(body);
if (!parsed.ok) return reply.status(400).send({ error: parsed.error }); // 在入口处把失败转成 400
const user = parsed.value; // 类型是 User，且数据已经过校验
```

**安全上要注意的地方：**
- 请求体完全由调用方控制。如果 `User` 里有 `id`、`role`、`tenantId`、`isAdmin` 这类字段，被伪造的后果就是越权。这类字段不能从 body 里读，要从认证后的身份和服务端状态推导出来。
- schema 里只放允许客户端提交的字段。用 `strictObject` 后，多出来的 `isAdmin` 会被直接拒绝。下面测试里有一条负向用例专门覆盖这一点。

## 3. 替换 `localStorage.getItem('cfg') as Config`

配置和请求体的策略不同：

- 读不到、存的内容坏了、或者结构不对，都**回退到默认值，并告诉调用方原因**。
- 多出来的字段直接去掉（`z.object`），新加的字段用 `.default` 补上，这样老版本存下来的配置也能读。

```ts
// src/config/config.ts
import { z } from 'zod';
import { parseWith } from '../boundary/parse';
import { getErrorMessage } from '../boundary/error';
import type { BoundaryError } from '../boundary/result';

export const CONFIG_KEY = 'cfg';

export const ConfigSchema = z.object({
  theme: z.enum(['light', 'dark']).default('light'),
  pageSize: z.number().int().min(1).max(200).default(20),
  features: z.array(z.string()).default(() => []), // 用工厂函数，避免多次读取共用同一个数组
});
export type Config = z.output<typeof ConfigSchema>;

export const defaultConfig = (): Config => ConfigSchema.parse({});

type FallbackReason = { kind: 'missing' } | { kind: 'unreadable'; message: string } | BoundaryError;
export type LoadedConfig =
  | { source: 'stored'; config: Config }
  | { source: 'default'; config: Config; reason: FallbackReason };

type ReadableStorage = Pick<Storage, 'getItem'>;

export function loadConfig(getStorage: () => ReadableStorage = () => window.localStorage): LoadedConfig {
  let raw: string | null;
  try {
    raw = getStorage().getItem(CONFIG_KEY); // Safari 隐私模式或禁用存储时会抛 SecurityError
  } catch (err) {
    return { source: 'default', config: defaultConfig(), reason: { kind: 'unreadable', message: getErrorMessage(err) } };
  }
  if (raw === null) return { source: 'default', config: defaultConfig(), reason: { kind: 'missing' } };

  const parsed = parseWith(ConfigSchema, raw);
  return parsed.ok
    ? { source: 'stored', config: parsed.value }
    : { source: 'default', config: defaultConfig(), reason: parsed.error };
}

export function saveConfig(config: Config, storage: Pick<Storage, 'setItem'> = window.localStorage): void {
  storage.setItem(CONFIG_KEY, JSON.stringify(config));
}
```

**安全上要注意：** 用户能通过开发者工具或 XSS 改写 localStorage。这里的配置只能影响界面偏好，**权限和计费开关不能放在这里**，必须由服务端判断。

## 5. 用编译器和 lint 卡住，防止再回退

`tsconfig.json`：`strict: true` 已经包含 `useUnknownInCatchVariables`。打开之后，所有直接用 `err.message` 的地方都会编译报错，正好用来把它们全部找出来。

```js

  '@typescript-eslint/no-unsafe-assignment': 'error',
  '@typescript-eslint/no-unsafe-member-access': 'error',
  '@typescript-eslint/no-unsafe-argument': 'error',
  '@typescript-eslint/no-unsafe-return': 'error',
```

迁移时，用下面几个正则搜出要改的地方，然后逐个替换：
- `JSON\.parse\([^)]*\)\s+as\s`
- `getItem\(`
- `catch\s*\(\s*\w+\s*:\s*any`
- `\.catch\(`

```ts
// src/user/user.test.ts
import { describe, expect, it } from 'vitest';
import { parseUser } from './user';

const valid = { id: 'u1', name: 'Ann', email: 'ann@example.com' };

describe('parseUser', () => {
  it('accepts a valid body', () => {
    expect(parseUser(JSON.stringify(valid))).toEqual({ ok: true, value: valid });
  });
  it('rejects malformed JSON', () => {
    const r = parseUser('{"id":');
    expect(r.ok).toBe(false);
    if (!r.ok) expect(r.error.kind).toBe('invalid-json');
  });
  it.each([
    ['missing email', { id: 'u1', name: 'Ann' }, 'email'],
    ['wrong type', { ...valid, age: '18' }, 'age'],
    ['bad email', { ...valid, email: 'nope' }, 'email'],
  ])('rejects %s', (_, body, path) => {
    const r = parseUser(JSON.stringify(body));
    expect(r.ok).toBe(false);
    if (!r.ok && r.error.kind === 'invalid-shape') {
      expect(r.error.issues.map((i) => i.path)).toContain(path);
    }
  });
  it.each(['null', '[]', '42', '"str"'])('rejects non-object top level %s', (body) => {
    expect(parseUser(body).ok).toBe(false);
  });
  // 负向安全用例：越权字段不能被带进来
  it('rejects privilege fields smuggled in the body', () => {
    expect(parseUser(JSON.stringify({ ...valid, isAdmin: true })).ok).toBe(false);
    expect(parseUser(`{"id":"u1","name":"Ann","email":"ann@example.com","__proto__":{"isAdmin":true}}`).ok).toBe(false);
  });
});
```

```ts
// src/config/config.test.ts
import { describe, expect, it } from 'vitest';
import { CONFIG_KEY, defaultConfig, loadConfig } from './config';

const storageWith = (value: string | null) => () => ({ getItem: (k: string) => (k === CONFIG_KEY ? value : null) });

describe('loadConfig', () => {
  it('missing → default', () => {
    expect(loadConfig(storageWith(null))).toEqual({ source: 'default', config: defaultConfig(), reason: { kind: 'missing' } });
  });
  it('corrupt JSON → default with reason', () => {
    const r = loadConfig(storageWith('{'));
    expect(r.source).toBe('default');
    if (r.source === 'default') expect(r.reason.kind).toBe('invalid-json');
  });
  it('wrong shape → default, never a half-typed object', () => {
    const r = loadConfig(storageWith('{"pageSize":"20"}'));
    expect(r).toMatchObject({ source: 'default', reason: { kind: 'invalid-shape' } });
    expect(r.config.pageSize).toBe(20);
  });
  it('partial old config gets new defaults, unknown keys stripped', () => {
    const r = loadConfig(storageWith('{"theme":"dark","legacy":1}'));
    expect(r).toEqual({ source: 'stored', config: { theme: 'dark', pageSize: 20, features: [] } });
  });
  it('storage throwing (private mode) → default', () => {
    const r = loadConfig(() => ({ getItem: () => { throw new DOMException('denied', 'SecurityError'); } }));
    expect(r).toMatchObject({ source: 'default', reason: { kind: 'unreadable', message: 'denied' } });
  });
  it('defaults are not shared between loads', () => {
    loadConfig(storageWith(null)).config.features.push('x');
    expect(loadConfig(storageWith(null)).config.features).toEqual([]);
  });
});
```

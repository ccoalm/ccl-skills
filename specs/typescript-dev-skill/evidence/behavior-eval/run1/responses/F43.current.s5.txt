## 思路

问题的根源是：`JSON.parse`、`localStorage` 和 `catch` 拿到的值，编译期都不知道是什么，而 `as User` 只是让编译器别管。改法是在这三个入口：

1. **入口一律当 `unknown`**：外部数据先是 `unknown`，必须经过运行时校验才能变成具体类型。
2. **用 schema 生成类型**：用 zod 写 schema，`User` 由 `z.infer` 得到，校验规则和类型不会各写一份、慢慢对不上。
3. **用工具防止回退**：靠 tsconfig 加 ESLint 禁止新代码再写 `JSON.parse(x) as T` 或 `err.message`。

## 1. 公共工具

```ts
// src/lib/errors.ts
function isRecord(v: unknown): v is Record<string, unknown> {
  return typeof v === 'object' && v !== null;
}

/** 把 catch 到的任意值转成 Error，自身永不抛错 */
export function toError(e: unknown): Error {
  if (e instanceof Error) return e;
  if (typeof e === 'string') return new Error(e);
  // 跨 iframe 的 Error，或第三方库 throw { message }
  if (isRecord(e) && typeof e.message === 'string') {
    return new Error(e.message, { cause: e });
  }
  let msg: string;
  try {
    msg = JSON.stringify(e) ?? String(e);
  } catch {
    // 循环引用、BigInt、Object.create(null) 等
    try { msg = String(e); } catch { msg = Object.prototype.toString.call(e); }
  }
  return new Error(msg, { cause: e });
}

export const errorMessage = (e: unknown): string => toError(e).message;
```

```ts
// src/lib/safe-json.ts
import { z } from 'zod';
import { toError } from './errors';

export type ParseResult<T> = { ok: true; value: T } | { ok: false; error: Error };

export class ValidationError extends Error {
  override name = 'ValidationError';
  constructor(readonly zodError: z.ZodError) {
    super(z.prettifyError(zodError));
  }
}

export function parseJson(text: string): ParseResult<unknown> {
  try {
    const value: unknown = JSON.parse(text); // 用注解收窄成 unknown，不用 as
    return { ok: true, value };
  } catch (e) {
    return { ok: false, error: toError(e) };
  }
}

export function parseJsonAs<S extends z.ZodType>(schema: S, text: string): ParseResult<z.output<S>> {
  const raw = parseJson(text);
  if (!raw.ok) return raw;
  const r = schema.safeParse(raw.value);
  return r.success
    ? { ok: true, value: r.data }
    : { ok: false, error: new ValidationError(r.error) };
}
```

## 3. 修改 `localStorage.getItem('cfg')`

```ts
// src/config/config.ts
import { z } from 'zod';
import { parseJsonAs } from '../lib/safe-json';

// 字段级 .catch：某一个字段坏了只回退这一个字段，不会丢掉整份配置
export const ConfigSchema = z.object({
  theme: z.enum(['light', 'dark']).catch('light'),
  pageSize: z.number().int().min(1).max(100).catch(20),
  locale: z.string().catch('zh-CN'),
});
export type Config = z.infer<typeof ConfigSchema>;

const KEY = 'cfg';
type StorageLike = Pick<Storage, 'getItem' | 'setItem'>;

export const defaultConfig = (): Config => ConfigSchema.parse({}); // 每次返回新对象，避免被共享修改

function defaultStorage(): StorageLike | undefined {
  try {
    return typeof localStorage === 'undefined' ? undefined : localStorage; // SSR / Node 环境
  } catch {
    return undefined; // Safari 隐私模式、禁用 cookie 时访问会抛 SecurityError
  }
}

export function loadConfig(storage: StorageLike | undefined = defaultStorage()): Config {
  let text: string | null;
  try {
    text = storage?.getItem(KEY) ?? null;
  } catch {
    return defaultConfig();
  }
  if (text === null) return defaultConfig();
  const r = parseJsonAs(ConfigSchema, text);
  return r.ok ? r.value : defaultConfig(); // JSON 损坏或根节点不是对象时，整体回退默认值
}

export function saveConfig(cfg: Config, storage: StorageLike | undefined = defaultStorage()): boolean {
  if (!storage) return false;
  try {
    storage.setItem(KEY, JSON.stringify(ConfigSchema.parse(cfg)));
    return true;
  } catch {
    return false; // 配额超限等情况
  }
}
```

`storage` 通过参数注入，测试时不需要 jsdom。

// src/models/user.test.ts
import { describe, it, expect, expectTypeOf } from 'vitest';
import { parseJson, parseJsonAs, ValidationError, type ParseResult } from '../lib/safe-json';
import { UserSchema, type User } from './user';

const valid = { id: 'u1', name: 'A', email: 'a@example.com' };

describe('parseJsonAs(UserSchema)', () => {
  it('合法输入', () => {
    const r = parseJsonAs(UserSchema, JSON.stringify(valid));
    expect(r).toEqual({ ok: true, value: valid });
  });
  it('非法 JSON → SyntaxError，不抛出', () => {
    const r = parseJsonAs(UserSchema, '{');
    expect(r.ok).toBe(false);
    if (!r.ok) expect(r.error.name).toBe('SyntaxError');
  });
  it('id 是 number（原来 as User 会放过的情况）', () => {
    const r = parseJsonAs(UserSchema, JSON.stringify({ ...valid, id: 1 }));
    expect(r.ok).toBe(false);
    if (!r.ok) {
      expect(r.error).toBeInstanceOf(ValidationError);
      expect((r.error as ValidationError).zodError.issues[0]?.path).toEqual(['id']);
    }
  });
  it.each(['null', '[]', '"str"', '{}'])('根节点 %s 被拒', (t) => {
    expect(parseJsonAs(UserSchema, t).ok).toBe(false);
  });
  it('多余字段和 __proto__ 被去掉', () => {
    const r = parseJsonAs(UserSchema, `{"id":"u1","name":"A","email":"a@example.com","role":"admin","__proto__":{"x":1}}`);
    expect(r.ok && Object.keys(r.value).sort()).toEqual(['email', 'id', 'name']);
    expect(({} as Record<string, unknown>).x).toBeUndefined();
  });
  it('类型层面', () => {
    expectTypeOf(parseJson('1')).toEqualTypeOf<ParseResult<unknown>>();
    const r = parseJsonAs(UserSchema, '{}');
    if (r.ok) expectTypeOf(r.value).toEqualTypeOf<User>();
    const raw = parseJson('{}');
    if (raw.ok) {
      // @ts-expect-error unknown 不能直接访问属性
      void raw.value.name;
    }
  });
});

// src/config/config.test.ts
import { describe, it, expect, vi } from 'vitest';
import { loadConfig, saveConfig, defaultConfig } from './config';

const mem = (init: Record<string, string> = {}) => {
  const m = new Map(Object.entries(init));
  return { getItem: (k: string) => m.get(k) ?? null, setItem: (k: string, v: string) => void m.set(k, v) };
};
const D = { theme: 'light', pageSize: 20, locale: 'zh-CN' };

describe('loadConfig', () => {
  it('无存储 / 无 key → 默认值', () => {
    expect(loadConfig(undefined)).toEqual(D);
    expect(loadConfig(mem())).toEqual(D);
  });
  it('合法配置', () => {
    const cfg = { theme: 'dark', pageSize: 50, locale: 'en' };
    expect(loadConfig(mem({ cfg: JSON.stringify(cfg) }))).toEqual(cfg);
  });
  it.each(['{', 'null', '[]', '"x"', '123'])('损坏值 %s → 默认值', (raw) => {
    expect(loadConfig(mem({ cfg: raw }))).toEqual(D);
  });
  it('单个字段非法只回退该字段', () => {
    expect(loadConfig(mem({ cfg: '{"theme":"neon","pageSize":50}' })))
      .toEqual({ ...D, pageSize: 50 });
  });
  it('pageSize 越界 / 类型错', () => {
    expect(loadConfig(mem({ cfg: '{"pageSize":0}' })).pageSize).toBe(20);
    expect(loadConfig(mem({ cfg: '{"pageSize":"50"}' })).pageSize).toBe(20);
  });
  it('getItem 抛 SecurityError → 默认值', () => {
    const s = { getItem: vi.fn(() => { throw new DOMException('denied', 'SecurityError'); }), setItem: vi.fn() };
    expect(loadConfig(s)).toEqual(D);
  });
  it('默认值不共享引用', () => {
    const a = loadConfig(undefined);
    a.pageSize = 99;
    expect(defaultConfig().pageSize).toBe(20);
  });
});

describe('saveConfig', () => {
  it('写入后可读回', () => {
    const s = mem();
    expect(saveConfig({ theme: 'dark', pageSize: 30, locale: 'en' }, s)).toBe(true);
    expect(loadConfig(s)).toEqual({ theme: 'dark', pageSize: 30, locale: 'en' });
  });
  it('配额超限返回 false 不抛', () => {
    const s = { getItem: () => null, setItem: () => { throw new DOMException('full', 'QuotaExceededError'); } };
    expect(saveConfig(defaultConfig(), s)).toBe(false);
  });
  it('无存储返回 false', () => expect(saveConfig(defaultConfig(), undefined)).toBe(false));
});
